cmake_minimum_required(VERSION 3.16)
project(bwr_display_additions LANGUAGES C)
set(CMAKE_C_STANDARD 11)
option(BWR_SANITIZERS "Enable host AddressSanitizer and UndefinedBehaviorSanitizer" ON)
if(BWR_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
enable_testing()
add_subdirectory(components/bwr_display)
add_executable(test_display tests/test_display.c)
target_link_libraries(test_display PRIVATE bwr_display)
add_test(NAME display_models COMMAND test_display)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME remote_control COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_remote.py)
add_test(NAME benchmark_app COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_app.py)

add_executable(test_raster tests/test_raster.c)
target_link_libraries(test_raster PRIVATE bwr_display)
add_test(NAME raster_equivalence COMMAND test_raster)

add_executable(test_band_scheduler tests/test_band_scheduler.c)
target_link_libraries(test_band_scheduler PRIVATE bwr_display)
add_test(NAME adaptive_band_scheduler COMMAND test_band_scheduler)

add_executable(test_video tests/test_video.c)
target_link_libraries(test_video PRIVATE bwr_display)
add_test(NAME video_codec COMMAND test_video)

add_test(NAME video_app COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_video_app.py)
add_test(NAME adaptive_video COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_adaptive_video.py)
add_test(NAME poster_video COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_poster.py)
add_test(NAME video_render_styles COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_light_grid.py)

add_test(NAME usb_host_framing COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_rpc.py)
add_test(NAME tcon_lab_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_tcon_lab.py)
add_test(NAME pingpong_optical_analysis COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_pingpong_analysis.py)

add_test(NAME adapter_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_adapter_contract.py)
